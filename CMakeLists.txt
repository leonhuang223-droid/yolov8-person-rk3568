cmake_minimum_required(VERSION 3.16)
project(person_rk3568 VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(PERSON_BUILD_APP "Build RKNN application (AArch64 runtime required)" ON)
option(PERSON_ENABLE_RGA "Compile RGA preprocessor" ON)
option(PERSON_BUILD_TESTS "Build portable logic tests" ON)
option(PERSON_USE_MOCK_RUNTIME "Host integration only: replace RKNN with a mock; no installation" OFF)
find_package(Threads REQUIRED)
add_library(person_core src/core.cpp)
target_include_directories(person_core PUBLIC include)
target_compile_options(person_core PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(person_core PUBLIC Threads::Threads)
if(PERSON_BUILD_APP)
  if(NOT OpenCV_DIR AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/opencv/lib/cmake/opencv4/OpenCVConfig.cmake")
    set(OpenCV_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/opencv/lib/cmake/opencv4" CACHE PATH "OpenCV configuration directory")
  endif()
  find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs videoio)
  set(RKNN_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/third_party/rknn" CACHE PATH "RKNN headers and lib directory")
  set(RGA_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/third_party/rga" CACHE PATH "RGA headers and lib directory")
  if(PERSON_USE_MOCK_RUNTIME)
    message(WARNING "TEST ONLY: this binary uses a mock Runtime and cannot measure NPU accuracy or speed")
    set(PERSON_ENABLE_RGA OFF)
    add_library(mock_rknn STATIC tests/mock_rknn.cpp)
    target_include_directories(mock_rknn PRIVATE "${RKNN_ROOT}/include")
    set(RKNN_LIBRARY mock_rknn)
  else()
    if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64")
      message(FATAL_ERROR "Bundled Runtime requires AArch64. Use cross toolchain, or PERSON_BUILD_APP=OFF for host tests")
    endif()
    find_library(RKNN_LIBRARY rknnrt PATHS "${RKNN_ROOT}/lib" NO_DEFAULT_PATH)
    if(NOT RKNN_LIBRARY)
      message(FATAL_ERROR "librknnrt.so missing: set RKNN_ROOT to a directory containing include/ and lib/")
    endif()
  endif()
  add_executable(person_detect src/main.cpp src/engine.cpp src/preprocess.cpp src/source.cpp src/output.cpp)
  target_include_directories(person_detect PRIVATE "${RKNN_ROOT}/include" ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(person_detect PRIVATE person_core ${OpenCV_LIBS} "${RKNN_LIBRARY}")
  target_compile_options(person_detect PRIVATE -Wall -Wextra -Wpedantic)
  if(PERSON_ENABLE_RGA)
    find_library(RGA_LIBRARY rga PATHS "${RGA_ROOT}/lib" NO_DEFAULT_PATH)
    if(NOT RGA_LIBRARY)
      message(FATAL_ERROR "librga.so missing: set RGA_ROOT or disable PERSON_ENABLE_RGA")
    endif()
    target_include_directories(person_detect SYSTEM PRIVATE "${RGA_ROOT}/include")
    target_link_libraries(person_detect PRIVATE "${RGA_LIBRARY}")
    target_compile_definitions(person_detect PRIVATE PERSON_ENABLE_RGA=1)
    install(FILES "${RGA_LIBRARY}" DESTINATION lib)
  endif()
  set_target_properties(person_detect PROPERTIES INSTALL_RPATH "$ORIGIN/../lib" BUILD_RPATH "${RKNN_ROOT}/lib;${RGA_ROOT}/lib")
  if(NOT PERSON_USE_MOCK_RUNTIME)
  install(TARGETS person_detect RUNTIME DESTINATION bin)
  install(FILES "${RKNN_LIBRARY}" DESTINATION lib)
  # Copy imported shared OpenCV modules with their SONAME symlink chains.
  foreach(opencv_module IN LISTS OpenCV_LIBS)
    if(TARGET "${opencv_module}")
      get_target_property(opencv_location "${opencv_module}" IMPORTED_LOCATION_RELEASE)
      if(NOT opencv_location)
        get_target_property(opencv_location "${opencv_module}" IMPORTED_LOCATION)
      endif()
      if(opencv_location MATCHES "\\.so(\\.|$)")
        get_filename_component(opencv_library_dir "${opencv_location}" DIRECTORY)
        if(EXISTS "${opencv_library_dir}/lib${opencv_module}.so")
          set(opencv_location "${opencv_library_dir}/lib${opencv_module}.so")
        endif()
        install(CODE "file(INSTALL DESTINATION \"\${CMAKE_INSTALL_PREFIX}/lib\" TYPE SHARED_LIBRARY FOLLOW_SYMLINK_CHAIN FILES \"${opencv_location}\")")
      endif()
    endif()
  endforeach()
  install(FILES model/best.rknn model/metadata.json DESTINATION model)
  install(FILES samples/test.jpg samples/test.mp4 DESTINATION .)
  install(FILES samples/video_metadata.json DESTINATION samples)
  install(FILES README.md DESTINATION .)
  install(DIRECTORY docs/ DESTINATION docs)
  install(FILES scripts/benchmark.py scripts/compare_jsonl.py scripts/check_jsonl.py scripts/check_package.py DESTINATION tools)
  install(PROGRAMS scripts/check_board.sh DESTINATION tools)
  install(DIRECTORY third_party/licenses/ DESTINATION licenses)
  install(FILES third_party/NOTICE.md DESTINATION licenses)
  endif()
endif()
if(PERSON_BUILD_TESTS)
  enable_testing()
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE person_core)
  add_test(NAME core_tests COMMAND core_tests)
  add_executable(pipeline_tests tests/pipeline_tests.cpp)
  target_link_libraries(pipeline_tests PRIVATE person_core)
  add_test(NAME pipeline_tests COMMAND pipeline_tests)
  set_tests_properties(pipeline_tests PROPERTIES TIMEOUT 30)
  add_executable(decode_dump tests/decode_dump.cpp)
  target_link_libraries(decode_dump PRIVATE person_core)
endif()
